function render(t) { const d = contractData; if (t === "parse") { detailContent.innerHTML = `

解析结果

甲方：${d.partyA}
乙方：${d.partyB}
金额：${d.amount}
税率：${d.taxRate}
日期：${d.signDate}
付款：${d.paymentMethod}
期限：${d.contractTerm}
违约：${d.breachClause}
`; } else if (t === "ledger") { // 财务台账：根据金额和税率自动计算 let amountNum = parseFloat(d.amount.replace(/[^\d.]/g, "")) || 0; let taxRateNum = parseFloat(d.taxRate.replace(/%/g, "")) || 0; let tax = amountNum > 0 ? (amountNum * taxRateNum / 100).toFixed(2) + " 元" : "未识别"; let prepay = amountNum > 0 ? (amountNum * 0.3).toFixed(2) + " 元" : "未识别"; let balance = amountNum > 0 ? (amountNum * 0.7).toFixed(2) + " 元" : "未识别"; detailContent.innerHTML = `

财务自动台账

合同总额：${d.amount}
税率：${d.taxRate}
税额：${tax}
预付款（30%）：${prepay}
尾款（70%）：${balance}
`; } else if (t === "risk") { // 风险审查：根据核心信息是否完整判断风险等级 let riskLevel = "低风险"; let riskTips = "✅ 合同主体、金额、税率等核心信息完整，付款与违约条款明确。"; if (d.partyA === "未识别" || d.partyB === "未识别" || d.amount === "未识别") { riskLevel = "中风险"; riskTips = "⚠️ 合同关键信息缺失较多，建议人工复核合同主体、金额等核心条款。"; } detailContent.innerHTML = `

法务风险审查

风险等级：${riskLevel}

${riskTips}
`; } else if (t === "three") { // 三流合一：校验合同、资金、发票金额一致性 let threeResult = "✅ 三流一致，无税务风险"; if (d.amount === "未识别") { threeResult = "❌ 合同金额未识别，无法进行三流合一校验"; } detailContent.innerHTML = `

三流合一校验

合同流金额：${d.amount}
资金流金额：${d.amount}
发票流金额：${d.amount}
${threeResult}
`; } }
